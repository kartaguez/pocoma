Demo is suspended/resumed through its existing Compose containers only. Its
persistent host files and stable SQL rows are compared without exporting values.
"""

import shutil

commands=[]

def save(name,data):
    (OUT/name).write_text(json.dumps(data,indent=2)+'\n')

def query(args,cwd=SOURCE):
    return subprocess.check_output([str(a) for a in args],cwd=cwd,text=True).strip()

def files(path):
    return {str(p.relative_to(path)):(hashlib.sha256(p.read_bytes()).hexdigest(),p.stat().st_mode & 0o777)
            for p in path.rglob('*') if p.is_file() and 'runs' not in p.relative_to(path).parts and p.name!='lock'}

def facade(action,phase):
    print('BEGIN '+phase,flush=True)
    began=time.time()
    result=subprocess.run(['./pocoma-local',action,'--instance',INSTANCE],cwd=SOURCE,
                          capture_output=True,text=True,timeout=1800)
    try:data=json.loads(result.stdout[result.stdout.index('{'):])
    except (ValueError,IndexError):
        # Retain raw failure privately only; it is not publication allowlisted.
        (BASE/(phase+'.private.txt')).write_text(result.stdout+result.stderr)
        raise RuntimeError('FACADE-OUTPUT-INVALID:'+phase)
    commands.append(dict(phase=phase,cwd=str(SOURCE),command=['./pocoma-local',action,'--instance',INSTANCE],
                         exit=result.returncode,seconds=time.time()-began,result=data))
    save('commands.json',commands)
    target=OUT/phase;target.mkdir()
    evidence=Path(data['evidence']) if data.get('evidence') else None
    if evidence:
        for name in ('receipt.json','checks.json','keycloak-auth.json','grafana.json','worker-recovery.json',
                     'generation-before.json','start-attempt.json','bruno-receipt.json','bruno.xml','scenarios.jsonl'):
            if (evidence/name).is_file():shutil.copyfile(evidence/name,target/name)
    print('END '+phase+': '+data['state']+' / '+str(result.returncode),flush=True)
    if result.returncode:raise RuntimeError('FACADE-FAILED:'+phase)
    return data

path=HISTORICAL/'.pocoma-local/instances/demo'
m=rt.load_manifest(path)
rt.ownership(m)

def demo_containers():
    result={}
    for resource in rt.compose(path,m,'ps','-aq').splitlines():
        row,=rt.docker_json('inspect',resource)
        result[row['Config']['Labels']['com.docker.compose.service']]=dict(id=row['Id'],state=row['State']['Status'],
            image=row['Image'],volumes=sorted(x['Name'] for x in row.get('Mounts',[]) if x['Type']=='volume'))
    return result

def stable_data():
    result={}
    for db,table in [('pocoma','keycloak.user_entity'),('pocoma','keycloak.credential'),('pocoma','keycloak.realm'),
                     ('pocoma','keycloak.client'),('pocoma_write','v3_write.v3_pot_identity'),
                     ('pocoma_read','v3_read.read_current_pot'),('pocoma_write','v3_write.v3_pot_event')]:
        result[table]=rt.sql(path,m,db,'SELECT md5(row_to_json(t)::text) FROM '+table+' t ORDER BY 1;').splitlines()
    result['sqlRoles']=rt.sql(path,m,'pocoma',"SELECT md5(string_agg(rolname || ':' || coalesce(rolpassword,''), E'\\n' ORDER BY rolname)) FROM pg_authid;").splitlines()
    return result

def demo_compose(args,phase):
    print('BEGIN '+phase,flush=True)
    began=time.time();rt.compose(path,m,*args,timeout=300)
    commands.append(dict(phase=phase,cwd=str(HISTORICAL),command=['docker','compose','-p',m['project'],'-f',str(path/'compose.json'),*args],exit=0,seconds=time.time()-began))
    save('commands.json',commands)
    print('END '+phase,flush=True)
