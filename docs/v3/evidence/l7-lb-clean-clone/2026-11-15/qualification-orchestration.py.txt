"""Qualification orchestration only; all product actions use the shipped facade.

Run with Python >=3.11 after a real origin clone. No reset/destroy/down/prune.
The historical checkout is used only for demo preservation, never as an input
to the clean-clone installation, build, sessions or Bruno collection.
"""
import hashlib
import json
from pathlib import Path
import shutil
import subprocess
import time

BASE = Path(__file__).resolve().parent
SOURCE = BASE / 'source'
HISTORICAL = Path('/Users/julien.guezennec/Dev/projects/pocoma')
OUT = BASE / 'evidence'
EXPECTED = '0a3246b9bc58bb813163caf22c6a2108c6f88db9'
COMMANDS = []

def save(name, data):
    (OUT / name).write_text(json.dumps(data, indent=2) + '\n')

def query(args, cwd=SOURCE):
    p = subprocess.run(args, cwd=cwd, capture_output=True, text=True, check=True)
    return p.stdout.strip()

def facade(root, action, instance, phase):
    argv = ['./pocoma-local', action, '--instance', instance]
    print('BEGIN ' + phase, flush=True)
    started = time.time()
    p = subprocess.run(argv, cwd=root, capture_output=True, text=True, timeout=1800)
    (OUT / (phase + '.stdout.txt')).write_text(p.stdout)
    (OUT / (phase + '.stderr.txt')).write_text(p.stderr)
    data = json.loads(p.stdout[p.stdout.index('{'):])
    COMMANDS.append(dict(phase=phase, cwd=str(root), argv=argv,
                         startedAt=started, seconds=time.time()-started, exit=p.returncode, result=data))
    save('commands.json', COMMANDS)
    evidence = Path(data['evidence'])
    target = OUT / phase
    target.mkdir(exist_ok=True)
    # Explicit public receipt allowlist. Never copy secrets/config, HTTP bodies,
    # canary-intent, seed state, container logs, npm output or command payloads.
    for name in ('receipt.json', 'checks.json', 'keycloak-auth.json', 'grafana.json',
                 'worker-recovery.json', 'generation-before.json', 'start-attempt.json',
                 'bruno-receipt.json', 'bruno.xml', 'scenarios.jsonl', 'tooling-refresh.json'):
        if (evidence / name).exists():
            shutil.copyfile(evidence / name, target / name)
    print('END ' + phase + ': ' + data['state'] + ', exit=' + str(p.returncode), flush=True)
    return p.returncode, data

def snapshot(name):
    # No environment, config or credentials from docker inspect.
    result = {}
    for key, args in [('containers', ['docker','ps','-a','--no-trunc','--format','{{json .}}']),
                      ('volumes', ['docker','volume','ls','--format','{{json .}}']),
                      ('networks', ['docker','network','ls','--no-trunc','--format','{{json .}}'])]:
        result[key] = [json.loads(line) for line in query(args).splitlines()]
    save(name, result)

assert query(['git','rev-parse','HEAD']) == EXPECTED
assert query(['git','branch','--show-current']) == 'feat/l7-lb-local-playground'
assert not query(['git','status','--porcelain=v1','--untracked-files=all'])
assert not query(['git','ls-files','--others','--ignored','--exclude-standard'])
save('clone-initial.json', dict(sha=EXPECTED, branch=query(['git','branch','--show-current']),
    remote=query(['git','remote','get-url','origin']), status='', ignored=[],
    cloneCommand=['git','clone','--single-branch','--branch','feat/l7-lb-local-playground',
                  'https://github.com/kartaguez/pocoma.git',str(SOURCE)],
    clone='Real network clone, not local clone/worktree; sandbox DNS retry required elevated execution',
    observedAt=time.time(), gitConfig=query(['git','config','--local','--list']),
    policySha256=hashlib.sha256((SOURCE/'docs/testing/Reactor_Verification_Policy.md').read_bytes()).hexdigest(),
    quickstartSha256=hashlib.sha256((SOURCE/'docs/v3/L7_Local_Playground_Quickstart.md').read_bytes()).hexdigest()))
snapshot('docker-initial-snapshot.json')
demo_touched = False
try:
    demo_touched = True
    code, _ = facade(HISTORICAL, 'stop', 'demo', 'demo-stop')
    if code:
        raise RuntimeError('demo graceful stop failed; qualification not started')
    code, start = facade(SOURCE, 'start', 'clean-clone', 'clean-start')
    snapshot('docker-after-start.json')
    if code:
        raise RuntimeError('clean-clone start failed')
    path = SOURCE / '.pocoma-local/instances/clean-clone'
    m = json.loads((path/'manifest.json').read_text())
    save('manifest-public.json', {k:m[k] for k in ('instance','owner','project','ports','sha','inputs','architecture','images','build','toolchain')})
    shutil.copyfile(path/'resources.json', OUT/'resources.json')
    if (Path(start['evidence'])/'package.log').exists():
        shutil.copyfile(Path(start['evidence'])/'package.log', OUT/'package.txt')
    code, _ = facade(SOURCE, 'gui', 'clean-clone', 'clean-gui-config')
    if code:
        raise RuntimeError('GUI public environment preparation failed')
    code, _ = facade(SOURCE, 'bruno', 'clean-clone', 'clean-bruno')
    if code:
        raise RuntimeError('Bruno proof failed')
    code, _ = facade(SOURCE, 'status', 'clean-clone', 'clean-status')
    if code:
        raise RuntimeError('fresh final clean-clone status failed')
finally:
    if (SOURCE/'.pocoma-local/instances/clean-clone/compose.json').exists():
        code, _ = facade(SOURCE, 'stop', 'clean-clone', 'clean-stop')
        snapshot('docker-after-clean-stop.json')
        if code:
            raise RuntimeError('clean-clone stop failed: demo restore requires investigation')
    if demo_touched:
        code, _ = facade(HISTORICAL, 'start', 'demo', 'demo-restored')
        snapshot('docker-final-snapshot.json')
        if code:
            raise RuntimeError('demo restore failed: preserve diagnostics')
save('clone-after-run.json', dict(sha=query(['git','rev-parse','HEAD']),
    status=query(['git','status','--porcelain=v1','--untracked-files=all']),
    observedAt=time.time(), note='Only generated ignored instance/build/npm/GUI data added by documented facade'))
print('QUALIFICATION CYCLE COMPLETE', flush=True)
