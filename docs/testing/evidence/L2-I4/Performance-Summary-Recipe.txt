def stats(values):
 values=sorted(values)
 return {'n':len(values),'median':statistics.median(values),'p95':values[max(0,(95*len(values)+99)//100-1)],'max':values[-1]}
configs={}
with tarfile.open(archive,'r|gz') as t:
 for m in t:
  if not m.isfile() or '/v3-i4-evidence/i4_p' not in m.name or not m.name.endswith('.txt'):continue
  locks=[];dispatch=None
  for raw in t.extractfile(m):
   line=raw.decode()
   match=re.search(r'LOCK duration physical=COMMITTED key=(\S+) .*lowerNs=(\d+) upperNs=(\d+)',line)
   if match:locks.append({'kind':'partition' if match[1]=='partition' else 'source','lower_ns':int(match[2]),'upper_ns':int(match[3])})
   match=re.search(r'COST actual harness single-platform-thread dispatchNs=(\d+)',line)
   if match:dispatch=int(match[1])
   prefix=re.match(r'P0([123]) (?:sample=(-?\d+) population=(\d+) W=(\d+) B=(\d+) cap=(\d+)|sample=(-?\d+) N=(\d+) B=(\d+)|COUNT N=(\d+) Q=(\d+) sample=(-?\d+)) ns=(\d+)',line)
   if not prefix:continue
   parts=prefix.groups()
   if parts[0]=='1':sample=int(parts[1]);params={'N':int(parts[2]),'W':int(parts[3]),'B':int(parts[4]),'cap':int(parts[5])}
   elif parts[0]=='2':sample=int(parts[6]);params={'N':int(parts[7]),'B':int(parts[8])}
   else:sample=int(parts[11]);params={'N':int(parts[9]),'Q':int(parts[10])}
   key=m.name+' '+json.dumps(params,sort_keys=True)
   row=configs.setdefault(key,{'trace':m.name,'params':params,'warmups':5,'warmup_indices':[],'measured':[]})
   if sample<0:row['warmup_indices'].append(sample)
   if sample>=0:
    row['measured'].append({'sample':sample,'ns':int(parts[12]),'locks':locks,'dispatch_ns':dispatch})
   locks=[];dispatch=None
